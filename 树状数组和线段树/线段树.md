```python
# 节点的下标：从 1 开始，如果你想改成从 0 开始，需要把左右儿子下标分别改成 node*2+1 和 node*2+2

class SegmentTree:
    def __init__(self, n: int, init_val=0):
        # 线段树维护一个长为 n 的数组（下标从 0 到 n-1），元素初始值为 init_val
        # init_val 可以是 list 或者 int
        # 如果是 int，那么创建一个 list
        if isinstance(init_val, int):
            init_val = [init_val] * n
        self._n = n
        self._tree = [0] * (2 << (n - 1).bit_length())
        self._build(init_val, 1, 0, n - 1)

    # 合并两个 val
    def _merge_val(self, a: int, b: int) -> int:
        return max(a, b)  # **根据题目修改**

    # 合并左右儿子的 val 到当前节点的 val
    def _maintain(self, node: int) -> None:
        self._tree[node] = self._merge_val(self._tree[node * 2], self._tree[node * 2 + 1])

    # 用 a 初始化线段树
    # 时间复杂度 O(n)
    def _build(self, a: List[int], node: int, l: int, r: int) -> None:
        if l == r:  # 叶子
            self._tree[node] = a[l]  # 初始化叶节点的值
            return
        m = (l + r) // 2
        self._build(a, node * 2, l, m)  # 初始化左子树
        self._build(a, node * 2 + 1, m + 1, r)  # 初始化右子树
        self._maintain(node)

    def _update(self, node: int, l: int, r: int, i: int, val: int) -> None:
        if l == r:  # 叶子（到达目标）
            # 如果想直接替换的话，可以写 self._tree[node] = val
            self._tree[node] = self._merge_val(self._tree[node], val)
            return
        m = (l + r) // 2
        if i <= m:  # i 在左子树
            self._update(node * 2, l, m, i, val)
        else:  # i 在右子树
            self._update(node * 2 + 1, m + 1, r, i, val)
        self._maintain(node)

    def _query(self, node: int, l: int, r: int, ql: int, qr: int) -> int:
        if ql <= l and r <= qr:  # 当前子树完全在 [ql, qr] 内
            return self._tree[node]
        m = (l + r) // 2
        if qr <= m:  # [ql, qr] 在左子树
            return self._query(node * 2, l, m, ql, qr)
        if ql > m:  # [ql, qr] 在右子树
            return self._query(node * 2 + 1, m + 1, r, ql, qr)
        l_res = self._query(node * 2, l, m, ql, qr)
        r_res = self._query(node * 2 + 1, m + 1, r, ql, qr)
        return self._merge_val(l_res, r_res)

    # 更新 a[i] 为 _merge_val(a[i], val)
    # 时间复杂度 O(log n)
    def update(self, i: int, val: int) -> None:
        self._update(1, 0, self._n - 1, i, val)

    # 返回用 _merge_val 合并所有 a[i] 的计算结果，其中 i 在闭区间 [ql, qr] 中
    # 时间复杂度 O(log n)
    def query(self, ql: int, qr: int) -> int:
        return self._query(1, 0, self._n - 1, ql, qr)

    # 获取 a[i] 的值
    # 时间复杂度 O(log n)
    def get(self, i: int) -> int:
        return self._query(1, 0, self._n - 1, i, i)
```

# [3161. 物块放置查询](https://leetcode.cn/problems/block-placement-queries/)
有一条无限长的数轴，原点在 0 处，沿着 x 轴 **正** 方向无限延伸。

给你一个二维数组 `queries` ，它包含两种操作：

1. 操作类型 1 ：`queries[i] = [1, x]` 。在距离原点 `x` 处建一个障碍物。数据保证当操作执行的时候，位置 `x` 处 **没有** 任何障碍物。
2. 操作类型 2 ：`queries[i] = [2, x, sz]` 。判断在数轴范围 `[0, x]` 内是否可以放置一个长度为 `sz` 的物块，这个物块需要 **完全** 放置在范围 `[0, x]` 内。如果物块与任何障碍物有重合，那么这个物块 **不能** 被放置，但物块可以与障碍物刚好接触。注意，你只是进行查询，并 **不是** 真的放置这个物块。每个查询都是相互独立的。

请你返回一个 boolean 数组`results` ，如果第 `i` 个操作类型 2 的操作你可以放置物块，那么 `results[i]` 为 `true` ，否则为 `false` 。

> 操作1 在x处放一个障碍物，假设x左右两侧的障碍物(left_obs & right_obs), 那么现在一共分成了两段: \[left_obs, x\] 和 \[x, right_obs\]. 只需要做两次单点更新 

> 操作2 我们需要判断在 `[0, x]` 内能否放下长度为 `sz` 的物块。此时最大可用空间由两部分组成：
1. **完全包含在 `[0, x]` 内部的完整空白区间**：它们的右端点必然 $\le x$。我们直接用线段树查询 `seg.query(0, x)` 即可得到这些完整区间的最大长度。
2. **尾部被切断的残缺空白区间**：即从 `x` 左侧最近的障碍物 `prev_obs` 到 `x` 本身这一段，长度为 `x - prev_obs`。
这两部分的最大值，就是整个 `[0, x]` 范围内能放下的**最大物块长度**：
$$\text{max\_gap} = \max(\text{seg.query}(0, x), x - \text{prev\_obs})$$

```python
class SegmentTree:
    def __init__(self, n:int):
        # n: 区间长度
        self.n = n
        self.tree = [0] * (2 * n)
    
    def update(self, idx:int, val:int) -> None:
        idx += self.n # 转换为叶子节点的下标
        self.tree[idx] = val
        # 更新父节点的值
        while idx > 1:
            idx //= 2
            self.tree[idx] = max(self.tree[idx*2], self.tree[idx*2+1])
    
    def query(self, left:int, right:int) -> int:
        # 查询区间 [left, right] 的最大值
        left += self.n
        right += self.n
        res = 0
        while left <= right:
            if left % 2 == 1: # 如果 left 是右子节点, 则它的父节点不包含在查询区间内, 可以直接使用 left 的值
                res = max(res, self.tree[left]) # update result
                left += 1
            if right % 2 == 0: # 如果 right 是左子节点, 则它的父节点不包含在查询区间内, 可以直接使用 right 的值
                res = max(res, self.tree[right]) # update result
                right -= 1
            left //= 2
            right //= 2
        return res
    
class Solution:
    def getResults(self, queries: List[List[int]]) -> List[bool]:
        MX = 50005
        obstacles = SortedList([0, MX]) # 动态维护障碍物的位置
        seg = SegmentTree(MX+1) # 线段树维护每个位置的最大高度
        seg.update(MX, MX) # 初始状态下，位置MX的高度为MX，表示边界
        res = []

        for q in queries:
            if q[0] == 1:
                # 添加障碍物
                x = q[1]
                # 找到x左右最近的障碍物位置
                i = obstacles.bisect_left(x)
                left_obstacle = obstacles[i-1]
                right_obstacle = obstacles[i]
                # 插入新的障碍物
                obstacles.add(x)
                # 更新被新障碍物分割的区间内的长度
                seg.update(x, x - left_obstacle) # 更新左侧区间的长度
                seg.update(right_obstacle, right_obstacle - x) # 更新右侧区间的长度
            elif q[0] == 2:
                x, sz = q[1], q[2]
                # 找到<=x的最大障碍物位置
                i = obstacles.bisect_right(x) - 1
                left_obstacle = obstacles[i]

                # 最大可用空间
                max_space = max(seg.query(0, x), x - left_obstacle)
                res.append(max_space >= sz)
        return res
```

# [2213. 由单个字符重复的最长子字符串](https://leetcode.cn/problems/longest-substring-of-one-repeating-character/)
给你一个下标从 **0** 开始的字符串 `s` 。另给你一个下标从 **0** 开始、长度为 `k` 的字符串 `queryCharacters` ，一个下标从 `0` 开始、长度也是 `k` 的整数 **下标** 数组 `queryIndices` ，这两个都用来描述 `k` 个查询。

第 `i` 个查询会将 `s` 中位于下标 `queryIndices[i]` 的字符更新为 `queryCharacters[i]` 。

返回一个长度为 `k` 的数组 `lengths` ，其中 `lengths[i]` 是在执行第 `i` 个查询 **之后** `s` 中仅由 **单个字符重复** 组成的 **最长子字符串** 的 **长度** _。_

```python
class SegmentTree2:
    def __init__(self, s: List[int]):
        self.n = len(s)
        self._s = s
        self.tree = [None] * (2 << (self.n-1).bit_length())
        self.build(s, 1, 0, self.n-1)
    
    def maintain(self, node:int, l:int, m:int, r:int):
        # mx:维护左半部分的最长连续子串长度
        # pre:维护左半部分的前缀长度    
        # suf:维护右半部分的后缀长度
        a_mx, a_pre, a_suf = self.tree[node*2] 
        b_mx, b_pre, b_suf = self.tree[node*2+1]
        same = self._s[m] == self._s[m+1] # left final char == right final char
        # 在合并区间时，a_pre表示左半部分的前缀长度，只有当左半步的前缀完全覆盖了左子区间本身(a_pre == m-l+1)
        # 且左右相邻字符相同时候 才能将左半部分的前缀长度与右半部分的前缀长度相加，得到新的前缀长度。同理，a_suf表示左半部分的后缀长度，只有当右半部分的后缀完全覆盖了右子区间本身(b_suf == r-m)
        self.tree[node] = (
            max(a_mx, b_mx, a_suf + b_pre if same else max(a_mx, b_mx)),
            a_pre + b_pre if same and a_pre == m-l+1 else a_pre,
            a_suf + b_suf if same and b_suf == r-m else b_suf
        )
    
    def build(self, s: List[int], node:int, l:int, r:int):
        if l == r:
            self.tree[node] = (1, 1, 1)
            return
        m = (l + r) // 2
        self.build(s, node*2, l, m)
        self.build(s, node*2+1, m+1, r)
        self.maintain(node, l, m, r)
    
    def _update(self, node:int, l:int, r:int, i:int, val:str):
        if l == r:
            self._s[i] = val
            return
        m = (l + r) // 2
        if i <= m:
            self._update(node*2, l, m, i, val)
        else:
            self._update(node*2+1, m+1, r, i, val)
        self.maintain(node, l, m, r)
    
    def update(self, i:int, val:str):
        self._update(1, 0, self.n-1, i, val)
    
    def query_all(self)->int:
        return self.tree[1][0]

class Solution:
    def longestRepeating(self, s: str, queryCharacters: str, queryIndices: List[int]) -> List[int]:
        t = SegmentTree2(list(s))
        ans = []
        for i, ch in zip(queryIndices, queryCharacters):
            t.update(i, ch)
            ans.append(t.query_all())
        return ans
```

# [3525. 求出数组的 X 值 II](https://leetcode.cn/problems/find-x-value-of-array-ii/)
给你一个由 **正整数** 组成的数组 `nums` 和一个 **正整数** `k`。同时给你一个二维数组 `queries`，其中 `queries[i] = [indexi, valuei, starti, xi]`。你可以对 `nums` 执行 **一次** 操作，移除 `nums` 的任意 **后缀** ，使得 `nums` 仍然**非空**。

给定一个 `x`，`nums` 的 **x值** 定义为执行以上操作后剩余元素的 **乘积** 除以 `k` 的 **余数** 为 `x` 的方案数。

对于 `queries` 中的每个查询，你需要执行以下操作，然后确定 `xi` 对应的 `nums` 的 **x值**：
- 将 `nums[indexi]` 更新为 `valuei`。仅这个更改在接下来的所有查询中保留。
- **移除** 前缀 `nums[0..(starti - 1)]`（`nums[0..(-1)]` 表示 **空前缀** ）。

返回一个长度为 `queries.length` 的数组 `result`，其中 `result[i]` 是第 `i` 个查询的答案。

数组的一个 **前缀** 是从数组开始位置到任意位置的子数组。

数组的一个 **后缀** 是从数组中任意位置开始直到结束的子数组。

**子数组** 是数组中一段连续的元素序列。

**注意**：操作中所选的前缀或后缀可以是 **空的** 。

**注意**：x值在本题中与问题 I 有不同的定义。



**思路**：

- 把子数组 $[l,r]$ 一分为二，左半下标范围为 $[l,m]$，右半下标范围为 $[m+1,r]$。其中 $m=⌊2l+r​⌋$
- 递归左半部分，计算left end为$l$, right end为$l,l+1,..., m$ 的子数组中元素模$k$ 等于 $x$的子数组一共有多少个
- 同理递归右半部份，计算left end在$m+1$, right end在$m+1,...,r$ 的子数组中元素模$k$ 等于 $x$的子数组一共有多少个
- 那么如果我们设左半的元素乘积模$k$的结果是$leftMult$, 那么把右半子数组的元素乘积$x\ leftMul$,就得到了左端点为$l$, 右端点在$m+1,...,r$的子数组的元素积，从而可以统计个数。
	- 假设右半有 $c=cnt[rx]$ 个乘积模 $k$ 等于 $rx$ 的子数组，那么在左端点为 $l$，右端点为 $m+1,m+2,…,r$ 的子数组中，有 $c$ 个元素积模 $k$ 等于 $leftMul⋅rx\mod k$ 的子数组。
	- 二者相加，就可以得到左端点为 $l$，右端点为 $m+1,m+2,…,r$的元素积模 $k$ 等于$x$的子数组的个数

转化成线段树的思想，线段树的每个节点维护一段子数组（区间），其中维护区间 $[l,r]$ 的节点保存：
- $mul$：整个区间 $[l,r]$ 的元素积模 $k$ 的结果。这可以用左右子树的 $mul$ 相乘模 $k$ 得到。
- cnt 数组：其中 $cnt[x]$ 表示在左端点为 $l$，右端点为 $l,l+1,l+2,…,r$ 的子数组中，元素积模 $k$ 等于 $x$ 的子数组的个数。计算规则就是上文的分治。

```python
class SegmentTree:
    def __init__(self, a: List[int], k:int):
        self.n = n = len(a)
        self._k = k
        self._tree = [None] * (2 << (n-1).bit_length())
        self._build(a, 1, 0, n-1)
    
    def _merge_data(self, a: Tuple[int, List[int]], b: Tuple[int, List[int]]):
        # 这里面的a和b是两个区间的统计信息，a[0]是左区间的乘积，a[1]是左区间的余数计数器，b[0]是右区间的乘积，b[1]是右区间的余数计数器
        cnt = a[1].copy()
        left_mul = a[0]
        for rx, c in enumerate(b[1]):
            cnt[left_mul * rx % self._k] += c
        return (left_mul * b[0] % self._k, cnt)
    
    def _new_data(self, val:int) -> Tuple[int, List[int]]:
        mul = val % self._k
        cnt = [0] * self._k
        cnt[mul] = 1
        return (mul, cnt)
    
    def _maintain(self, node:int) -> None:
        self._tree[node] = self._merge_data(self._tree[node*2], self._tree[node*2+1])
    
    def _build(self, a: List[int], node:int, l:int, r:int) -> None:
        if l == r:
            self._tree[node] = self._new_data(a[l]) # 初始化叶子节点
            return
        mid = (l + r) // 2
        self._build(a, node*2, l, mid)
        self._build(a, node*2+1, mid+1, r)
        self._maintain(node)
    
    def _update(self, node:int, l:int, r:int, idx:int, val:int) -> None:
        if l == r:
            self._tree[node] = self._new_data(val)
            return
        mid = (l + r) // 2
        if idx <= mid:
            self._update(node*2, l, mid, idx, val)
        else:
            self._update(node*2+1, mid+1, r, idx, val)
        self._maintain(node)
    
    def _query(self, node:int, l:int, r:int, ql:int, qr:int) -> Tuple[int, List[int]]:
        if ql <= l and r <= qr: # 完全覆盖
            return self._tree[node]
        mid = (l + r) // 2
        if qr <= mid:
            return self._query(node*2, l, mid, ql, qr)
        elif ql > mid:
            return self._query(node*2+1, mid+1, r, ql, qr)
        left_data = self._query(node*2, l, mid, ql, qr)
        right_data = self._query(node*2+1, mid+1, r, ql, qr)
        return self._merge_data(left_data, right_data)
    
    def update(self, idx:int, val:int) -> None:
        # 更新 a[idx] = val
        self._update(node=1, l=0, r=self.n-1, idx=idx, val=val)
    
    def query(self, ql:int, qr:int) -> Tuple[int, List[int]]:
        # 返回用 _merge_data 合并所有 a[i] 的计算结果，其中 i 在闭区间 [ql, qr] 中
        return self._query(node=1, l=0, r=self.n-1, ql=ql, qr=qr)

class Solution:
    def resultArray(self, nums: List[int], k: int, queries: List[List[int]]) -> List[int]:
        t = SegmentTree(nums, k)
        res = []
        n = len(nums)
        for index, val, start, x in queries:
            t.update(index, val)
            mul, cnt = t.query(start, n-1)
            res.append(cnt[x])
        return res
```
